### github下载文件
	- `./publish.sh URL`
	- ```
	  #!/bin/bash
	  set -euo pipefail
	  
	  # ============================================================
	  # 通用 GitHub Workspace 预制品发布脚本（自动命名版）
	  # 用法: ./publish.sh <下载URL>
	  # 示例: ./publish.sh https://dist.neo4j.org/neo4j-community-2026.07.1-windows.zip
	  # ============================================================
	  
	  # --- 配置 ---
	  DOWNLOAD_URL="${1:-}"          # 唯一参数：下载链接
	  REPO="${GITHUB_REPOSITORY:-}"  # 自动获取当前仓库
	  
	  # 如果没有提供 URL，提示用法
	  if [ -z "$DOWNLOAD_URL" ]; then
	      echo "❌ 用法: $0 <下载URL>"
	      echo "   示例:"
	      echo "     $0 https://dist.neo4j.org/neo4j-community-2026.07.1-windows.zip"
	      echo "     $0 https://example.com/some/file.tar.gz"
	      exit 1
	  fi
	  
	  # --- 自动命名规则 ---
	  # 1. 从 URL 提取 basename（去掉查询参数）
	  BASENAME="$(basename "$DOWNLOAD_URL" | sed 's/[?#].*$//')"
	  
	  # 2. 如果 basename 为空或只有域名路径（如以 / 结尾），自动生成
	  if [ -z "$BASENAME" ] || [[ "$BASENAME" == "/" ]] || [[ "$BASENAME" == *"/"* ]]; then
	      TIMESTAMP=$(date +%s)
	      BASENAME="download-${TIMESTAMP}.bin"
	      echo "⚠️  URL 中未检测到文件名，使用自动命名: $BASENAME"
	  fi
	  
	  # 3. 最终文件名（保留原始文件名，无需修改）
	  FILENAME="$BASENAME"
	  
	  # 确定工作目录
	  WORKSPACE="${GITHUB_WORKSPACE:-$(pwd)}"
	  FILEPATH="$WORKSPACE/$FILENAME"
	  
	  echo "========================================"
	  echo "📦 Workspace 预制品发布工具"
	  echo "========================================"
	  echo "🔹 仓库:     ${REPO:-未知}"
	  echo "🔹 下载地址: $DOWNLOAD_URL"
	  echo "🔹 自动命名: $FILENAME"
	  echo ""
	  
	  # --- Step 1: 下载文件 ---
	  echo "⬇️  正在下载..."
	  if command -v curl &>/dev/null; then
	      curl -fsSL --progress-bar -o "$FILEPATH" "$DOWNLOAD_URL"
	  elif command -v wget &>/dev/null; then
	      wget --progress=bar:force -O "$FILEPATH" "$DOWNLOAD_URL"
	  else
	      echo "❌ 需要 curl 或 wget"
	      exit 1
	  fi
	  
	  FILESIZE=$(du -h "$FILEPATH" | cut -f1)
	  echo "✅ 下载完成: $FILENAME ($FILESIZE)"
	  echo ""
	  
	  # --- Step 2: 检查 Token ---
	  GH_TOKEN="${GH_TOKEN:-${GITHUB_TOKEN:-}}"
	  if [ -z "$GH_TOKEN" ]; then
	      echo "❌ 未找到 GH_TOKEN 或 GITHUB_TOKEN"
	      exit 1
	  fi
	  export GH_TOKEN
	  
	  # --- Step 3: 创建 Prerelease ---
	  TIMESTAMP=$(date +%s)
	  TAG_NAME="prerelease-${TIMESTAMP}"
	  
	  echo "🚀 正在发布 Prerelease: $TAG_NAME"
	  
	  # 如果标签已存在则删除（防止冲突）
	  if [ -n "$REPO" ]; then
	      gh api --method DELETE "repos/$REPO/git/refs/tags/$TAG_NAME" &>/dev/null || true
	  fi
	  
	  gh release create "$TAG_NAME" \
	      "$FILEPATH" \
	      --repo "${REPO:-}" \
	      --prerelease \
	      --title "📦 Prerelease ${TIMESTAMP}" \
	      --notes "自动发布的预制品
	  
	  - **文件**: \`$FILENAME\`
	  - **大小**: $FILESIZE
	  - **时间**: $(date '+%Y-%m-%d %H:%M:%S')
	  - **来源**: <$DOWNLOAD_URL>"
	  
	  echo ""
	  echo "========================================"
	  echo "✅ 发布成功！"
	  echo "========================================"
	  
	  # --- Step 4: 生成下载链接 ---
	  if [ -n "$REPO" ]; then
	      DOWNLOAD_LINK="https://github.com/$REPO/releases/download/$TAG_NAME/$FILENAME"
	      RELEASE_LINK="https://github.com/$REPO/releases/tag/$TAG_NAME"
	  
	      echo ""
	      echo "🔗 Release 页面: $RELEASE_LINK"
	      echo "🔗 直接下载:     $DOWNLOAD_LINK"
	      echo ""
	  
	      # 尝试自动打开浏览器（Codespaces/本地支持）
	      if command -v xdg-open &>/dev/null; then
	          echo "🌐 正在尝试打开浏览器..."
	          xdg-open "$DOWNLOAD_LINK" &>/dev/null || true
	      elif command -v open &>/dev/null; then
	          echo "🌐 正在尝试打开浏览器..."
	          open "$DOWNLOAD_LINK" &>/dev/null || true
	      elif [ -n "${CODESPACES:-}" ]; then
	          echo "💡 在 Codespaces 中，你可以直接点击上面的链接下载"
	      fi
	  
	      # 输出 Markdown 格式，方便复制
	      echo ""
	      echo "📋 Markdown 引用:"
	      echo "[$FILENAME]($DOWNLOAD_LINK)"
	  else
	      echo "⚠️  无法生成下载链接（未检测到仓库信息）"
	  fi
	  
	  echo ""
	  echo "🎉 完成！"
	  ```
